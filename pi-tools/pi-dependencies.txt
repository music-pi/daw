# System packages required for maschinepi on Raspberry Pi
# These will be installed via apt-get inside the chroot

# Audio system (PipeWire)
pipewire
pipewire-alsa
pipewire-jack
pipewire-pulse
wireplumber
libpipewire-0.3-dev

# ALSA (for compatibility)
libasound2

# Build tools (for compiling maschinepi and mk3-boot-display)
build-essential
cmake
git
pkg-config
ninja-build

# Runtime dependencies
libc6-dev
libstdc++6

# System utilities
systemd
udev
